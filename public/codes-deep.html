* CPU Scheduling

FCFS (First-Come, First-Served)

FCFS is the simplest scheduling algorithm where the process that arrives first is executed first. It calculates the waiting time and turnaround time for a fixed set of burst times. 
#include <stdio.h>
                
int main() {
    int n = 4, bt[] = {6, 8, 7, 3};
    int wt[4], tat[4];
    float total_wt = 0, total_tat = 0;

    // Waiting time for first process is 0
    wt[0] = 0;

    // Calculate waiting time
    for (int i = 1; i < n; i++) {
        wt[i] = bt[i-1] + wt[i-1];
    }

    // Calculate turnaround time and totals
    for (int i = 0; i < n; i++) {
        tat[i] = bt[i] + wt[i];
        total_wt += wt[i];
        total_tat += tat[i];
    }

    printf("Process\tBurst Time\tWaiting Time\tTurnaround Time\n");
    for (int i = 0; i < n; i++) {
        printf("%d\t%d\t\t%d\t\t%d\n", i+1, bt[i], wt[i], tat[i]);
    }
    printf("Average Waiting Time: %.2f\n", total_wt / n);
    printf("Average Turnaround Time: %.2f\n", total_tat / n);
    return 0;
}

SJF (Shortest Job First - Non-Preemptive)

Non-preemptive SJF selects the process with the shortest burst time next. This implementation sorts the processes by burst time and calculates wait/turnaround times. 
#include <stdio.h>

int main() {
    int p[] = {1, 2, 3, 4};
    int bt[] = {8, 4, 6, 3};
    int wt[4], tat[4];
    float total_wt = 0, total_tat = 0;
    int n = 4, temp;

    // Sort processes by burst time
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n - i - 1; j++) {
            if (bt[j] > bt[j + 1]) {
                temp = bt[j]; bt[j] = bt[j + 1]; bt[j + 1] = temp;
                temp = p[j]; p[j] = p[j + 1]; p[j + 1] = temp;
            }
        }
    }

    // Calculate times
    wt[0] = 0;
    for (int i = 1; i < n; i++) {
        wt[i] = bt[i-1] + wt[i-1];
        total_wt += wt[i];
    }
    for (int i = 0; i < n; i++) {
        tat[i] = bt[i] + wt[i];
        total_tat += tat[i];
    }

    printf("Process\tBurst Time\tWaiting Time\tTurnaround Time\n");
    for (int i = 0; i < n; i++) {
        printf("%d\t%d\t\t%d\t\t%d\n", p[i], bt[i], wt[i], tat[i]);
    }
    printf("Average Waiting Time: %.2f\n", total_wt / n);
    return 0;
}

Priority Scheduling (Non-Preemptive)

In Priority scheduling, processes are executed based on their assigned priority. This example assumes lower number = higher priority. It sorts the processes by priority number. 
#include <stdio.h>

int main() {
    int p[] = {1, 2, 3, 4};
    int bt[] = {8, 4, 6, 3};
    int pr[] = {2, 3, 1, 4}; // Lower number = Higher priority
    int wt[4], tat[4];
    float total_wt = 0, total_tat = 0;
    int n = 4, temp;

    // Sort by priority
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n - i - 1; j++) {
            if (pr[j] > pr[j + 1]) {
                temp = pr[j]; pr[j] = pr[j + 1]; pr[j + 1] = temp;
                temp = bt[j]; bt[j] = bt[j + 1]; bt[j + 1] = temp;
                temp = p[j]; p[j] = p[j + 1]; p[j + 1] = temp;
            }
        }
    }

    wt[0] = 0;
    for (int i = 1; i < n; i++) {
        wt[i] = bt[i-1] + wt[i-1];
        total_wt += wt[i];
    }
    for (int i = 0; i < n; i++) {
        tat[i] = bt[i] + wt[i];
        total_tat += tat[i];
    }

    printf("Process\tBurst Time\tPriority\tWaiting Time\tTurnaround Time\n");
    for (int i = 0; i < n; i++) {
        printf("%d\t%d\t\t%d\t\t%d\t\t%d\n", p[i], bt[i], pr[i], wt[i], tat[i]);
    }
    return 0;
}

Round Robin (RR)

Round Robin assigns a fixed time quantum to each process and cycles through them. This implementation maintains a circular queue and updates the remaining burst times until all processes finish. 
#include <stdio.h>
#define MAX 100

int main() {
    int n = 4, quantum = 2;
    int bt[] = {6, 8, 7, 3};
    int rt[4];
    int wt[4] = {0}, tat[4];
    float total_wt = 0, total_tat = 0;
    int queue[MAX], front = 0, rear = 0, current = 0;
    int visited[4] = {0};

    // Initialize remaining time and queue
    for (int i = 0; i < n; i++) {
        rt[i] = bt[i];
        queue[rear++] = i;
    }

    // Round Robin Loop
    while (front < rear) {
        int p = queue[front++];
        if (rt[p] > quantum) {
            rt[p] -= quantum;
            current += quantum;
            // If not in queue, add to rear
            queue[rear++] = p;
        } else {
            current += rt[p];
            rt[p] = 0;
            tat[p] = current;
            wt[p] = tat[p] - bt[p];
            total_wt += wt[p];
            total_tat += tat[p];
        }
    }

    printf("Process\tBurst Time\tWaiting Time\tTurnaround Time\n");
    for (int i = 0; i < n; i++) {
        printf("%d\t%d\t\t%d\t\t%d\n", i+1, bt[i], wt[i], tat[i]);
    }
    printf("Average Turnaround Time: %.2f\n", total_tat / n);
    return 0;
}

SRJF (Shortest Remaining Job First - Preemptive)

SRJF is the preemptive version of SJF where a process can be interrupted if a new process arrives with a shorter remaining time. 
#include <stdio.h>
#include <limits.h>

int main() {
    int n = 4;
    int at[] = {0, 1, 2, 3};
    int bt[] = {8, 4, 9, 5};
    int rt[4], wt[4], tat[4];
    int complete = 0, t = 0, minm = INT_MAX, shortest = 0, finish_time;
    float total_wt = 0, total_tat = 0;
    
    for (int i = 0; i < n; i++) rt[i] = bt[i];
    
    // Process until all processes complete
    while (complete != n) {
        minm = INT_MAX;
        for (int i = 0; i < n; i++) {
            if (at[i] <= t && rt[i] < minm && rt[i] > 0) {
                minm = rt[i];
                shortest = i;
            }
        }
        if (minm == INT_MAX) {
            t++;
            continue;
        }
        rt[shortest]--;
        t++;
        if (rt[shortest] == 0) {
            complete++;
            finish_time = t;
            tat[shortest] = finish_time - at[shortest];
            wt[shortest] = tat[shortest] - bt[shortest];
            total_wt += wt[shortest];
            total_tat += tat[shortest];
        }
    }

    printf("Process\tBurst Time\tWaiting Time\tTurnaround Time\n");
    for (int i = 0; i < n; i++) {
        printf("%d\t%d\t\t%d\t\t%d\n", i+1, bt[i], wt[i], tat[i]);
    }
    printf("Average Waiting Time: %.2f\n", total_wt / n);
    return 0;
}

* Page Replacement

FIFO (First-In-First-Out) Page Replacement

When a page fault occurs, FIFO removes the page that has been in the memory frames for the longest time. 
#include <stdio.h>
#include <stdbool.h>

int main() {
    int pages[] = {7, 0, 1, 2, 0, 3, 0, 4, 2, 3, 0, 3, 2, 1, 2, 0, 1, 7, 0, 1};
    int n = 20, frames = 3, frame[3];
    int front = 0, page_faults = 0;
    
    for (int i = 0; i < frames; i++) frame[i] = -1;
    
    for (int i = 0; i < n; i++) {
        bool found = false;
        for (int j = 0; j < frames; j++) {
            if (frame[j] == pages[i]) {
                found = true;
                break;
            }
        }
        if (!found) {
            frame[front] = pages[i];
            front = (front + 1) % frames;
            page_faults++;
        }
        printf("Frame: ");
        for (int j = 0; j < frames; j++) {
            if (frame[j] != -1) printf("%d ", frame[j]);
            else printf("- ");
        }
        printf("\n");
    }
    printf("Total Page Faults: %d\n", page_faults);
    return 0;
}

Optimal Page Replacement

The optimal page replacement algorithm replaces the page that will not be used for the longest period of time in the future. 
#include <stdio.h>
#include <stdbool.h>
#include <limits.h>

int findOptimal(int pages[], int n, int frame[], int frames, int current) {
    int res = -1, farthest = -1;
    for (int i = 0; i < frames; i++) {
        int j;
        for (j = current + 1; j < n; j++) {
            if (frame[i] == pages[j]) {
                if (j > farthest) {
                    farthest = j;
                    res = i;
                }
                break;
            }
        }
        if (j == n) return i; // This page won't be used again
    }
    return (res == -1) ? 0 : res;
}

int main() {
    int pages[] = {7, 0, 1, 2, 0, 3, 0, 4, 2, 3, 0, 3, 2, 1, 2, 0, 1, 7, 0, 1};
    int n = 20, frames = 3, frame[3];
    int page_faults = 0;
    
    for (int i = 0; i < frames; i++) frame[i] = -1;
    
    for (int i = 0; i < n; i++) {
        bool found = false;
        for (int j = 0; j < frames; j++) {
            if (frame[j] == pages[i]) {
                found = true; break;
            }
        }
        if (!found) {
            if (i < frames) {
                frame[i] = pages[i];
            } else {
                int replace_idx = findOptimal(pages, n, frame, frames, i);
                frame[replace_idx] = pages[i];
            }
            page_faults++;
        }
        printf("Frame: ");
        for (int j = 0; j < frames; j++) {
            if (frame[j] != -1) printf("%d ", frame[j]);
            else printf("- ");
        }
        printf("\n");
    }
    printf("Total Page Faults: %d\n", page_faults);
    return 0;
}

LRU (Least Recently Used) Page Replacement

LRU replaces the page that has not been used for the longest period of time. It uses a timestamp or counter to track page usage. 
#include <stdio.h>
#include <stdbool.h>
#include <limits.h>

int findLRU(int time[], int frames) {
    int min = time[0], pos = 0;
    for (int i = 1; i < frames; i++) {
        if (time[i] < min) {
            min = time[i];
            pos = i;
        }
    }
    return pos;
}

int main() {
    int pages[] = {7, 0, 1, 2, 0, 3, 0, 4, 2, 3, 0, 3, 2, 1, 2, 0, 1, 7, 0, 1};
    int n = 20, frames = 3, frame[3];
    int time[3];
    int page_faults = 0, counter = 0;
    
    for (int i = 0; i < frames; i++) { frame[i] = -1; time[i] = 0; }

    for (int i = 0; i < n; i++) {
        bool found = false;
        for (int j = 0; j < frames; j++) {
            if (frame[j] == pages[i]) {
                found = true;
                time[j] = ++counter;
                break;
            }
        }
        if (!found) {
            if (i < frames) {
                frame[i] = pages[i];
                time[i] = ++counter;
            } else {
                int replace_idx = findLRU(time, frames);
                frame[replace_idx] = pages[i];
                time[replace_idx] = ++counter;
            }
            page_faults++;
        }
        printf("Frame: ");
        for (int j = 0; j < frames; j++) {
            if (frame[j] != -1) printf("%d ", frame[j]);
            else printf("- ");
        }
        printf("\n");
    }
    printf("Total Page Faults: %d\n", page_faults);
    return 0;
}

MRU (Most Recently Used) Page Replacement

MRU replaces the page that has been used most recently. It uses a timestamp, and on a page fault, it replaces the frame with the maximum timestamp. 
#include <stdio.h>
#include <stdbool.h>
#include <limits.h>

int findMRU(int time[], int frames) {
    int max = time[0], pos = 0;
    for (int i = 1; i < frames; i++) {
        if (time[i] > max) {
            max = time[i];
            pos = i;
        }
    }
    return pos;
}

* Fit Algorithms (Memory Allocation)

Best Fit

Best Fit searches the entire list of available memory blocks and allocates the smallest block that is large enough for the process. 
#include <stdio.h>
#include <limits.h>

void bestFit(int blockSize[], int m, int processSize[], int n) {
    int allocation[n];
    for (int i = 0; i < n; i++) allocation[i] = -1;
    
    for (int i = 0; i < n; i++) {
        int bestIdx = -1;
        for (int j = 0; j < m; j++) {
            if (blockSize[j] >= processSize[i]) {
                if (bestIdx == -1 || blockSize[j] < blockSize[bestIdx]) {
                    bestIdx = j;
                }
            }
        }
        if (bestIdx != -1) {
            allocation[i] = bestIdx;
            blockSize[bestIdx] -= processSize[i];
        }
    }
    
    printf("\nProcess No.\tProcess Size\tBlock no.\n");
    for (int i = 0; i < n; i++) {
        printf("%d\t\t%d\t\t", i+1, processSize[i]);
        if (allocation[i] != -1) printf("%d\n", allocation[i] + 1);
        else printf("Not Allocated\n");
    }
}

int main() {
    int blockSize[] = {100, 500, 200, 300, 600};
    int processSize[] = {212, 417, 112, 426};
    int m = 5, n = 4;
    bestFit(blockSize, m, processSize, n);
    return 0;
}

First Fit

First Fit allocates a process to the first available memory block that is large enough. It is faster than Best Fit but may waste more memory.

Banker's Algorithm (Deadlock Avoidance)

The Banker's Algorithm checks for a safe state to avoid deadlock. It calculates the Need matrix, runs the Safety Algorithm, and finds a safe execution sequence if one exists. 
#include <stdio.h>
#include <stdbool.h>
#define P 5 // Number of processes
#define R 3 // Number of resources

int main() {
    int avail[] = {3, 3, 2};
    int max[P][R] = {{7, 5, 3}, {3, 2, 2}, {9, 0, 2}, {2, 2, 2}, {4, 3, 3}};
    int alloc[P][R] = {{0, 1, 0}, {2, 0, 0}, {3, 0, 2}, {2, 1, 1}, {0, 0, 2}};
    int need[P][R];
    bool finish[P] = {false};
    int safeSeq[P];
    int count = 0;
    
    // Calculate Need Matrix
    for (int i = 0; i < P; i++) {
        for (int j = 0; j < R; j++) {
            need[i][j] = max[i][j] - alloc[i][j];
        }
    }
    
    // Safety Algorithm
    while (count < P) {
        bool found = false;
        for (int p = 0; p < P; p++) {
            if (!finish[p]) {
                bool can_alloc = true;
                for (int j = 0; j < R; j++) {
                    if (need[p][j] > avail[j]) {
                        can_alloc = false;
                        break;
                    }
                }
                if (can_alloc) {
                    for (int k = 0; k < R; k++) {
                        avail[k] += alloc[p][k];
                    }
                    safeSeq[count++] = p;
                    finish[p] = true;
                    found = true;
                }
            }
        }
        if (!found) {
            printf("System is not in a safe state!\n");
            return 0;
        }
    }
    
    printf("System is in a safe state.\nSafe Sequence: ");
    for (int i = 0; i < P; i++) {
        printf("P%d ", safeSeq[i]);
    }
    printf("\n");
    return 0;
}
