PROGRAM 01

Second

Identifies selected keywords, integers, and alphabetic words from the input. Other characters are echoed unchanged.

● second.l
/* A lex program to identify some keywords, numbers and words */
%{
#include<stdio.h>
%}

%%
if|else|printf {printf("%s is a keyword\n", yytext);}
[0-9]+ {printf("%s is a number\n", yytext);}
[a-zA-Z]+ {printf("%s is a word\n", yytext);}
.|\n {ECHO;}
%%

int main()
{
    printf("Enter your input string: ");
    yylex();
}

int yywrap()
{
    return 1;
}

PROGRAM 04

Sixsth

Checks whether a sentence contains selected conjunctions such as and, or, because, but, if, then, or nevertheless.

● sixsth.l
/* a program to check if a sentence is compund or not. */

%{
#include<stdio.h>
int isCompund = 0;
%}

%%
and|or|because|but|if|then|nevertheless {isCompund=1;}
. ;
\n {return 0;}
%%

int main()
{
    printf("Enter Your Sentence: ");
    yylex();
    if(isCompund == 1)  printf("The Sentence is a compund sentence.\n");
    else printf("The sentence is not a compund sentence\n");
}
int yywrap()
{
    return 1;
}
PROGRAM 05

Seventh

Counts lines, whitespace spaces/tabs, words, and total characters using yyleng.

● seventh.l
/* Cunting number of lines, words, spaces and characters in a sentenc */ 
%{
#include<stdio.h>
int line_count = 0, word_count = 0, space_count = 0, character_count = 0;
%}

%%
\n              { line_count++; character_count += yyleng; }
[ \t]           { space_count++; character_count += yyleng; }
[^ \t\n]+       { word_count++; character_count += yyleng; }
%%

int main()
{
    printf("Enter your sentenc: ");
    yylex();
    printf("Number of lines: %d\n", line_count);
    printf("Number of words: %d\n", word_count);
    printf("Number of space: %d\n", space_count);
    printf("Number of characters: %d\n", character_count);

}

int yywrap()
{
    return 1;
}

PROGRAM 07

Ninth

Checks a simple arithmetic expression made from numbers and the + and * operators, then lists the numbers and operators found.

● ninth.l
/* Recognize a valid arithmetic expression and recognize the identifiers only numbers and operators only + 
and * present */

%{
#include<stdio.h>
#include<string.h>
int identifier_count = 0, operator_count = 0, k, flag=0;
char identifiers[10][10], operators[10][10];
%}

%%
[0-9]+ {flag++; strcpy(identifiers[identifier_count], yytext); identifier_count++;}
[+*] {flag--; strcpy(operators[operator_count], yytext); operator_count++;}
%%

int main()
{
    printf("Enter the arithmetic expression: ");
    yylex();
    if(flag != 1) printf("Invalid Expression\n");
    else {
        printf("\nIdentifiers: \n");
        for(k = 0; k<identifier_count; k++){
            printf("%s\n", identifiers[k]);
        }

        printf("\nOperators: \n");
        for(k = 0; k<operator_count; k++){
            printf("%s\n", operators[k]);
        }
    }
}

int yywrap()
{
    return 1;
}
