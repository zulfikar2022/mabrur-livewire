OS Algorithms in C

CPU Scheduling

1. FCFS (First Come First Serve)

The simplest scheduling algorithm that executes processes in the exact order of their arrival. It is non-preemptive and can suffer from the "convoy effect" where short processes wait behind long ones. 

#include <stdio.h>

int main() {
    int burst_time[] = {24, 3, 3};
    int n = sizeof(burst_time)/sizeof(burst_time[0]);
    int wait_time[n], tat[n];
    
    wait_time[0] = 0;
    for (int i = 1; i < n; i++) {
        wait_time[i] = burst_time[i-1] + wait_time[i-1];
    }
    
    printf("Process  Burst Time  Wait Time  Turnaround Time\n");
    for (int i = 0; i < n; i++) {
        tat[i] = burst_time[i] + wait_time[i];
        printf("P%d\t\t%d\t\t%d\t\t%d\n", i+1, burst_time[i], wait_time[i], tat[i]);
    }
    return 0;
}

2. SJF (Shortest Job First - Non-Preemptive)

Selects the waiting process with the smallest execution time to execute next. It minimizes the average waiting time for all processes. 

#include <stdio.h>

int main() {
    int bt[] = {6, 8, 7, 3}, p[] = {1, 2, 3, 4}, n = 4;
    int wt[4], tat[4], temp;
    
    // Sort by Burst Time
    for(int i = 0; i < n; i++) {
        for(int j = i+1; j < n; j++) {
            if(bt[i] > bt[j]) {
                temp = bt[i]; bt[i] = bt[j]; bt[j] = temp;
                temp = p[i]; p[i] = p[j]; p[j] = temp;
            }
        }
    }
    
    wt[0] = 0;
    for(int i = 1; i < n; i++) wt[i] = bt[i-1] + wt[i-1];
    for(int i = 0; i < n; i++) tat[i] = bt[i] + wt[i];
    
    printf("Process  Burst Time  Wait Time\n");
    for(int i = 0; i < n; i++) 
        printf("P%d\t\t%d\t\t%d\n", p[i], bt[i], wt[i]);
        
    return 0;
}

3. Priority Scheduling

Each process is assigned a priority, and the CPU is allocated to the process with the highest priority (often represented by the lowest integer). 

#include <stdio.h>

int main() {
    int bt[] = {10, 1, 2, 1, 5}, pr[] = {3, 1, 4, 5, 2};
    int p[] = {1, 2, 3, 4, 5}, n = 5, wt[5], temp;
    
    // Sort by Priority (Lower number = higher priority)
    for(int i = 0; i < n; i++) {
        for(int j = i+1; j < n; j++) {
            if(pr[i] > pr[j]) {
                temp = pr[i]; pr[i] = pr[j]; pr[j] = temp;
                temp = bt[i]; bt[i] = bt[j]; bt[j] = temp;
                temp = p[i]; p[i] = p[j]; p[j] = temp;
            }
        }
    }
    
    wt[0] = 0;
    for(int i = 1; i < n; i++) wt[i] = bt[i-1] + wt[i-1];
    
    printf("Process  Priority  Burst Time  Wait Time\n");
    for(int i = 0; i < n; i++) 
        printf("P%d\t\t%d\t\t%d\t\t%d\n", p[i], pr[i], bt[i], wt[i]);
        
    return 0;
}

4. Round Robin

A preemptive algorithm designed for time-sharing systems. Each process gets a small unit of CPU time (time quantum) in turn. 

#include <stdio.h>
#include <stdbool.h>

int main() {
    int n = 3, quantum = 2;
    int bt[] = {10, 5, 8}, rem_bt[] = {10, 5, 8}, wt[3], tat[3];
    int t = 0; // Current time
    
    while(1) {
        bool done = true;
        for (int i = 0; i < n; i++) {
            if (rem_bt[i] > 0) {
                done = false;
                if (rem_bt[i] > quantum) {
                    t += quantum;
                    rem_bt[i] -= quantum;
                } else {
                    t = t + rem_bt[i];
                    wt[i] = t - bt[i];
                    rem_bt[i] = 0;
                }
            }
        }
        if (done == true) break;
    }
    
    printf("Process  Burst Time  Wait Time\n");
    for (int i = 0; i < n; i++)
        printf("P%d\t\t%d\t\t%d\n", i+1, bt[i], wt[i]);
        
    return 0;
}

5. SRJF (Shortest Remaining Job First)

The preemptive version of SJF. If a new process arrives with a shorter burst time than the remaining time of the currently executing process, the CPU is preempted. 

#include <stdio.h>
#include <stdbool.h>

int main() {
    int bt[] = {6, 8, 7, 3}, art[] = {0, 1, 2, 3}, n = 4;
    int rt[4], wt[4];
    for (int i = 0; i < n; i++) rt[i] = bt[i];
    
    int complete = 0, t = 0, minm = 10000;
    int shortest = 0, finish_time;
    bool check = false;

    while (complete != n) {
        for (int j = 0; j < n; j++) {
            if ((art[j] <= t) && (rt[j] < minm) && rt[j] > 0) {
                minm = rt[j];
                shortest = j;
                check = true;
            }
        }
        if (check == false) { t++; continue; }
        
        rt[shortest]--;
        minm = rt[shortest];
        if (minm == 0) minm = 10000;
        
        if (rt[shortest] == 0) {
            complete++;
            check = false;
            finish_time = t + 1;
            wt[shortest] = finish_time - bt[shortest] - art[shortest];
            if (wt[shortest] < 0) wt[shortest] = 0;
        }
        t++;
    }
    
    printf("Process  Wait Time\n");
    for (int i = 0; i < n; i++) printf("P%d\t\t%d\n", i+1, wt[i]);
    return 0;
}

Page Replacement

6. FIFO (First In First Out)

The simplest page replacement algorithm. It keeps track of all pages in memory in a queue, replacing the oldest page when a page fault occurs. 

#include <stdio.h>

int main() {
    int pages[] = {1, 3, 0, 3, 5, 6, 3};
    int n = 7, capacity = 3;
    int frames[3] = {-1, -1, -1};
    int faults = 0, pointer = 0;

    for (int i = 0; i < n; i++) {
        int found = 0;
        for (int j = 0; j < capacity; j++) {
            if (frames[j] == pages[i]) {
                found = 1;
                break;
            }
        }
        if (!found) {
            frames[pointer] = pages[i];
            pointer = (pointer + 1) % capacity;
            faults++;
        }
    }
    printf("Total Page Faults (FIFO): %d\n", faults);
    return 0;
}

7. Optimal Page Replacement

Replaces the page that will not be used for the longest period of time in the future. It has the lowest page fault rate but is impossible to implement in practice (used as a benchmark). 

#include <stdio.h>

int main() {
    int pages[] = {7, 0, 1, 2, 0, 3, 0, 4, 2, 3};
    int n = 10, capacity = 3, frames[3] = {-1, -1, -1}, faults = 0;

    for (int i = 0; i < n; i++) {
        int found = 0;
        for (int j = 0; j < capacity; j++) {
            if (frames[j] == pages[i]) { found = 1; break; }
        }
        
        if (!found) {
            int replace_idx = -1, farthest = i;
            // Check for empty frames first
            for(int j = 0; j < capacity; j++) {
                if(frames[j] == -1) { replace_idx = j; break; }
            }
            // If full, find the optimal frame to replace
            if(replace_idx == -1) {
                for (int j = 0; j < capacity; j++) {
                    int k;
                    for (k = i + 1; k < n; k++) {
                        if (frames[j] == pages[k]) {
                            if (k > farthest) { farthest = k; replace_idx = j; }
                            break;
                        }
                    }
                    if (k == n) { replace_idx = j; break; } // Page not used in future
                }
            }
            if(replace_idx == -1) replace_idx = 0;
            frames[replace_idx] = pages[i];
            faults++;
        }
    }
    printf("Total Page Faults (Optimal): %d\n", faults);
    return 0;
}

8. LRU (Least Recently Used)

Replaces the page that has not been used for the longest period of time. It approximates Optimal algorithm by looking at the past instead of the future. 

#include <stdio.h>

int main() {
    int pages[] = {7, 0, 1, 2, 0, 3, 0, 4, 2, 3};
    int n = 10, capacity = 3;
    int frames[3] = {-1, -1, -1}, last_used[3] = {0, 0, 0};
    int faults = 0, time = 0;

    for (int i = 0; i < n; i++) {
        time++;
        int found = 0;
        for (int j = 0; j < capacity; j++) {
            if (frames[j] == pages[i]) {
                found = 1;
                last_used[j] = time;
                break;
            }
        }
        if (!found) {
            int lru_idx = 0;
            int min_time = last_used[0];
            for (int j = 1; j < capacity; j++) {
                if (last_used[j] < min_time) {
                    min_time = last_used[j];
                    lru_idx = j;
                }
            }
            frames[lru_idx] = pages[i];
            last_used[lru_idx] = time;
            faults++;
        }
    }
    printf("Total Page Faults (LRU): %d\n", faults);
    return 0;
}

9. MRU (Most Recently Used)

Replaces the page that was most recently accessed. Often useful in situations where the older pages are more likely to be accessed again. 

#include <stdio.h>

int main() {
    int pages[] = {7, 0, 1, 2, 0, 3, 0, 4, 2, 3};
    int n = 10, capacity = 3;
    int frames[3] = {-1, -1, -1}, last_used[3] = {0, 0, 0};
    int faults = 0, time = 0;

    for (int i = 0; i < n; i++) {
        time++;
        int found = 0;
        for (int j = 0; j < capacity; j++) {
            if (frames[j] == pages[i]) {
                found = 1;
                last_used[j] = time;
                break;
            }
        }
        if (!found) {
            int mru_idx = 0;
            int has_empty = 0;
            // Check empty first
            for(int j = 0; j < capacity; j++) {
                if(frames[j] == -1) { mru_idx = j; has_empty = 1; break; }
            }
            // Replace MRU if full
            if(!has_empty) {
                int max_time = last_used[0];
                for (int j = 1; j < capacity; j++) {
                    if (last_used[j] > max_time) {
                        max_time = last_used[j];
                        mru_idx = j;
                    }
                }
            }
            frames[mru_idx] = pages[i];
            last_used[mru_idx] = time;
            faults++;
        }
    }
    printf("Total Page Faults (MRU): %d\n", faults);
    return 0;
}

Fit Algorithms

10. Best Fit

Allocates the smallest hole that is big enough for the process. It searches the entire list and helps save larger empty blocks for larger processes. 

#include <stdio.h>

int main() {
    int block[] = {100, 500, 200, 300, 600};
    int process[] = {212, 417, 112, 426};
    int m = 5, n = 4, allocation[4];
    
    for(int i = 0; i < n; i++) allocation[i] = -1;

    for (int i = 0; i < n; i++) {
        int best_idx = -1;
        for (int j = 0; j < m; j++) {
            if (block[j] >= process[i]) {
                if (best_idx == -1 || block[j] < block[best_idx]) {
                    best_idx = j;
                }
            }
        }
        if (best_idx != -1) {
            allocation[i] = best_idx;
            block[best_idx] -= process[i];
        }
    }

    printf("Process No. Process Size Block no.\n");
    for (int i = 0; i < n; i++) {
        printf("%d \t\t %d \t\t ", i+1, process[i]);
        if (allocation[i] != -1) printf("%d\n", allocation[i] + 1);
        else printf("Not Allocated\n");
    }
    return 0;
}

11. First Fit

Allocates the first hole in memory that is big enough for the process. It is generally the fastest allocation method. 

#include <stdio.h>

int main() {
    int block[] = {100, 500, 200, 300, 600};
    int process[] = {212, 417, 112, 426};
    int m = 5, n = 4, allocation[4];
    
    for(int i = 0; i < n; i++) allocation[i] = -1;

    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            if (block[j] >= process[i]) {
                allocation[i] = j;
                block[j] -= process[i];
                break;
            }
        }
    }

    printf("Process No. Process Size Block no.\n");
    for (int i = 0; i < n; i++) {
        printf("%d \t\t %d \t\t ", i+1, process[i]);
        if (allocation[i] != -1) printf("%d\n", allocation[i] + 1);
        else printf("Not Allocated\n");
    }
    return 0;
}

12. Worst Fit

Allocates the largest hole available in memory. The idea is to leave large leftover holes that can accommodate other processes later. 

#include <stdio.h>

int main() {
    int block[] = {100, 500, 200, 300, 600};
    int process[] = {212, 417, 112, 426};
    int m = 5, n = 4, allocation[4];
    
    for(int i = 0; i < n; i++) allocation[i] = -1;

    for (int i = 0; i < n; i++) {
        int worst_idx = -1;
        for (int j = 0; j < m; j++) {
            if (block[j] >= process[i]) {
                if (worst_idx == -1 || block[j] > block[worst_idx]) {
                    worst_idx = j;
                }
            }
        }
        if (worst_idx != -1) {
            allocation[i] = worst_idx;
            block[worst_idx] -= process[i];
        }
    }

    printf("Process No. Process Size Block no.\n");
    for (int i = 0; i < n; i++) {
        printf("%d \t\t %d \t\t ", i+1, process[i]);
        if (allocation[i] != -1) printf("%d\n", allocation[i] + 1);
        else printf("Not Allocated\n");
    }
    return 0;
}

Other Algorithms

13. Banker's Algorithm

A resource allocation and deadlock avoidance algorithm that tests for safety by simulating the allocation of predetermined maximum possible amounts of all resources. 

#include <stdio.h>

int main() {
    int n = 5; // Number of processes
    int m = 3; // Number of resources
    
    int alloc[5][3] = { {0, 1, 0}, {2, 0, 0}, {3, 0, 2}, {2, 1, 1}, {0, 0, 2} };
    int max[5][3] = { {7, 5, 3}, {3, 2, 2}, {9, 0, 2}, {2, 2, 2}, {4, 3, 3} };
    int avail[3] = {3, 3, 2}; 

    int f[5] = {0}, ans[5], ind = 0, need[5][3];

    for (int i = 0; i < n; i++) {
        for (int j = 0; j < m; j++) {
            need[i][j] = max[i][j] - alloc[i][j];
        }
    }
    
    for (int k = 0; k < n; k++) {
        for (int i = 0; i < n; i++) {
            if (f[i] == 0) {
                int flag = 0;
                for (int j = 0; j < m; j++) {
                    if (need[i][j] > avail[j]){
                        flag = 1;
                        break;
                    }
                }
                if (flag == 0) {
                    ans[ind++] = i;
                    for (int y = 0; y < m; y++)
                        avail[y] += alloc[i][y];
                    f[i] = 1;
                }
            }
        }
    }
    
    int flag = 1;
    for(int i=0; i < n; i++) {
        if(f[i] == 0) { flag = 0; printf("System is not safe\n"); break; }
    }
    
    if(flag == 1) {
        printf("Safe Sequence: ");
        for (int i = 0; i < n - 1; i++) printf("P%d -> ", ans[i]);
        printf("P%d\n", ans[n - 1]);
    }
    return 0;
}
